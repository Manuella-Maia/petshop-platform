-- =====================================================
-- 1. CRIAÇÃO E SELEÇÃO DO BANCO DE DADOS
-- =====================================================

-- Consulta os bancos de dados existentes
SHOW DATABASES;

-- Cria o banco de dados caso ainda não exista
CREATE DATABASE IF NOT EXISTS api_petshop;

-- Seleciona o banco que será utilizado
USE api_petshop;


-- =====================================================
-- 2. CRIAÇÃO DAS TABELAS
-- =====================================================

-- -----------------------------------------------------
-- Tabela: perfis
-- Armazena os tipos de acesso disponíveis no sistema.
-- -----------------------------------------------------

CREATE TABLE perfis (
    id_perfil INT AUTO_INCREMENT PRIMARY KEY,
    nome_perfil VARCHAR(30) NOT NULL,
    descricao_perfil VARCHAR(50) NOT NULL
);


-- -----------------------------------------------------
-- Tabela: usuarios
-- Armazena as credenciais e informações de acesso.
-- Cada usuário está associado a um perfil.
-- -----------------------------------------------------

CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    id_perfil INT NOT NULL,
    email_usuario VARCHAR(120) NOT NULL,
    senha_hash VARCHAR(255) NOT NULL,
    status_usuario VARCHAR(20) NOT NULL,
    bloqueado_usuario BOOLEAN NOT NULL,
    criado_usuario DATETIME NOT NULL,
    atualizado_usuario DATETIME NULL,

    FOREIGN KEY (id_perfil)
        REFERENCES perfis(id_perfil)
);


-- -----------------------------------------------------
-- Tabela: clientes
-- Armazena os dados pessoais dos clientes do pet shop.
-- Cada cliente possui um usuário de acesso.
-- -----------------------------------------------------

CREATE TABLE clientes (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    nome_cliente VARCHAR(80) NOT NULL,
    cpf_cliente VARCHAR(14) NOT NULL,
    cpf_hash_cliente VARCHAR(255) NOT NULL,
    telefone_cliente VARCHAR(15) NOT NULL,
    criado_cliente DATETIME NOT NULL,
    atualizado_cliente DATETIME NULL,

    FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
);


-- -----------------------------------------------------
-- Tabela: funcionarios
-- Armazena os dados dos funcionários da empresa.
-- Cada funcionário possui um usuário de acesso.
-- -----------------------------------------------------

CREATE TABLE funcionarios (
    id_funcionario INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    nome_funcionario VARCHAR(80) NOT NULL,
    email_funcionario VARCHAR(120) NOT NULL,
    cpf_funcionario VARCHAR(14) NOT NULL,
    cpf_hash_funcionario VARCHAR(255) NOT NULL,
    telefone_funcionario VARCHAR(15) NOT NULL,
    cargo_funcionario VARCHAR(20) NOT NULL,
    data_admissao DATE NOT NULL,
    data_desligamento DATE NULL,
    criado_funcionario DATETIME NOT NULL,
    atualizado_funcionario DATETIME NULL,

    FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
);


-- -----------------------------------------------------
-- Tabela: pets
-- Armazena os animais cadastrados no pet shop.
-- Cada animal está associado a um cliente.
-- -----------------------------------------------------

CREATE TABLE pets (
    id_pet INT AUTO_INCREMENT PRIMARY KEY,
    id_cliente INT NOT NULL,
    nome_pet VARCHAR(80) NOT NULL,
    especie_pet VARCHAR(50) NOT NULL,
    raca_pet VARCHAR(40) NOT NULL,
    porte VARCHAR(7) NOT NULL,
    peso_pet DECIMAL(10,2) NOT NULL,
    data_nascimento_pet DATE NOT NULL,
    observacao_pet VARCHAR(80) NULL,
    ativo_pet BOOLEAN NOT NULL,
    criado_pet DATETIME NOT NULL,
    atualizado_pet DATETIME NULL,

    FOREIGN KEY (id_cliente)
        REFERENCES clientes(id_cliente)
);


-- -----------------------------------------------------
-- Tabela: servicos
-- Armazena os serviços oferecidos pelo pet shop.
-- -----------------------------------------------------

CREATE TABLE servicos (
    id_servico INT AUTO_INCREMENT PRIMARY KEY,
    tipo_servico VARCHAR(20) NOT NULL,
    descricao VARCHAR(50) NULL,
    preco_base DECIMAL(10,2) NOT NULL,
    ativo_servico BOOLEAN NOT NULL,
    duracao TIME NOT NULL,
    criado_servico DATETIME NOT NULL,
    atualizado_servico DATETIME NULL
);


-- -----------------------------------------------------
-- Tabela: agendamentos
-- Registra os agendamentos dos animais.
-- Cada agendamento está associado a um pet e funcionário.
-- -----------------------------------------------------

CREATE TABLE agendamentos (
    id_agendamento INT AUTO_INCREMENT PRIMARY KEY,
    id_pet INT NOT NULL,
    id_funcionario INT NOT NULL,
    data_agendamento DATETIME NOT NULL,
    status_agendamento VARCHAR(20) NOT NULL,
    criado_agendamento DATETIME NOT NULL,
    atualizado_agendamento DATETIME NULL,

    FOREIGN KEY (id_pet)
        REFERENCES pets(id_pet),

    FOREIGN KEY (id_funcionario)
        REFERENCES funcionarios(id_funcionario)
);


-- -----------------------------------------------------
-- Tabela: agendamento_servicos
-- Relaciona os agendamentos aos serviços contratados.
-- Permite associar serviços aos agendamentos.
-- -----------------------------------------------------

CREATE TABLE agendamento_servicos (
    id_agendamento_servico INT AUTO_INCREMENT PRIMARY KEY,
    id_agendamento INT NOT NULL,
    id_servico INT NOT NULL,
    quantidade_servico INT NOT NULL,
    preco_unitario DECIMAL(10,2) NOT NULL,

    FOREIGN KEY (id_agendamento)
        REFERENCES agendamentos(id_agendamento),

    FOREIGN KEY (id_servico)
        REFERENCES servicos(id_servico)
);


-- -----------------------------------------------------
-- Tabela: auditoria
-- Registra as ações realizadas pelos usuários no sistema.
-- Cada registro de auditoria está associado a um usuário.
-- -----------------------------------------------------

CREATE TABLE auditoria (
    id_auditoria INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    acao VARCHAR(50) NOT NULL,
    valor_anterior VARCHAR(50) NOT NULL,
    valor_atualizado VARCHAR(50) NOT NULL,
    data_time_auditoria DATETIME NOT NULL,

    FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
);

-- =====================================================
-- 4. CRIAÇÃO DA VIEW
-- =====================================================

-- Cria uma visão para consultar os animais cadastrados.
-- A VIEW permite consultar os dados sem escrever o SELECT
-- completo novamente.

CREATE OR REPLACE VIEW vw_animais AS
SELECT * FROM pets;


-- =====================================================
-- 5. CRIAÇÃO DA PROCEDURE
-- =====================================================

-- Procedure responsável por cadastrar novos animais.
-- Recebe os dados do animal e o ID do cliente responsável.
-- Os campos ativo_pet e criado_pet são preenchidos
-- automaticamente durante a inserção.

DROP PROCEDURE IF EXISTS cadastrar_pet;

DELIMITER //

CREATE PROCEDURE cadastrar_pet(
    IN p_id_cliente INT,
    IN p_nome_pet VARCHAR(80),
    IN p_especie_pet VARCHAR(50),
    IN p_raca_pet VARCHAR(40),
    IN p_porte VARCHAR(7),
    IN p_peso_pet DECIMAL(10,2),
    IN p_data_nascimento_pet DATE,
    IN p_observacao_pet VARCHAR(80)
)
BEGIN

    INSERT INTO pets (
        id_cliente,
        nome_pet,
        especie_pet,
        raca_pet,
        porte,
        peso_pet,
        data_nascimento_pet,
        observacao_pet,
        ativo_pet,
        criado_pet
    )
    VALUES (
        p_id_cliente,
        p_nome_pet,
        p_especie_pet,
        p_raca_pet,
        p_porte,
        p_peso_pet,
        p_data_nascimento_pet,
        p_observacao_pet,
        TRUE,
        NOW()
    );

END //

DELIMITER ;
